   140→   140→   140→function getInitials(name: string): string {
   141→   141→   141→  if (!name) return "?";
   142→   142→   142→  const parts = name.trim().split(/\s+/);
   143→   143→   143→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
   144→   144→   144→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
   145→   145→   145→}

   154→   154→   154→}

   228→   228→   228→            size="sm"
   229→   229→   229→          >
   230→   230→   230→            <UserPlus className="h-4 w-4" />
   231→   231→   231→            <span className="hidden sm:inline">Novo cliente</span>
   232→   232→   232→            <span className="sm:hidden">Novo</span>
   233→   233→   233→          </Button>
   234→   234→   234→        }
   235→   235→   235→      />

   237→   237→   237→      {/* Search bar */}

   239→   239→   239→        <div className="relative">
   240→   240→   240→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   241→   241→   241→          <Input
   242→   242→   242→            value={query}
   243→   243→   243→            onChange={(e) => setQuery(e.target.value)}
   244→   244→   244→            placeholder="Buscar por nome, telefone, CPF/CNPJ ou e-mail..."
   245→   245→   245→            className="h-10 rounded-xl border-border bg-background/50 pl-9"
   246→   246→   246→          />
   247→   247→   247→        </div>
   248→   248→   248→        {!loading && (
   249→   249→   249→          <p className="mt-2 text-xs text-muted-foreground">
   250→   250→   250→            {customers.length} cliente{customers.length === 1 ? "" : "s"}
   251→   251→   251→            {debouncedQuery ? ` para "${debouncedQuery}"` : ""}
   252→   252→   252→          </p>
   253→   253→   253→        )}
   254→   254→   254→      </SectionCard>
   255→   255→   255→
   256→   256→   256→      {/* Customer list */}

   258→   258→   258→        <SectionCard>
   259→   259→   259→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">

   261→   261→   261→              <CustomerCardSkeleton key={i} />
   262→   262→   262→            ))}
   263→   263→   263→          </div>
   264→   264→   264→        </SectionCard>
   265→   265→   265→      ) : customers.length === 0 ? (
   266→   266→   266→        <SectionCard>
   267→   267→   267→          <EmptyState
   268→   268→   268→            icon={<Users className="h-10 w-10 opacity-40" />}
   269→   269→   269→            title={debouncedQuery ? "Nenhum cliente encontrado" : "Nenhum cliente cadastrado"}
   270→   270→   270→            description={
   271→   271→   271→              debouncedQuery
   272→   272→   272→                ? "Tente outro termo de busca."
   273→   273→   273→                : "Comece cadastrando seu primeiro cliente."
   274→   274→   274→            }
   275→   275→   275→            action={
   276→   276→   276→              !debouncedQuery && (
   277→   277→   277→                <Button
   278→   278→   278→                  onClick={() => setCreateOpen(true)}
   279→   279→   279→                  className="gradient-orange text-primary-foreground glow-orange"
   280→   280→   280→                  size="sm"
   281→   281→   281→                >

   315→   315→   315→                </div>

   317→   317→   317→                <div className="grid grid-cols-2 gap-2 text-[11px]">
