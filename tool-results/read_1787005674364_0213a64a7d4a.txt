   137→// ============================================================================

   140→function getInitials(name: string): string {
   141→  if (!name) return "?";
   142→  const parts = name.trim().split(/\s+/);
   143→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
   144→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
   145→}
   146→
   147→function maskPlate(plate?: string | null): string {
   148→  return (plate || "-").toUpperCase();
   149→}
   150→
   151→function formatMileage(km: number): string {
   152→  if (!km) return "0 km";
   153→  return `${km.toLocaleString("pt-BR")} km`;
   154→}

   166→
   167→  // Debounce search input
   168→  useEffect(() => {
   169→    const t = setTimeout(() => setDebouncedQuery(query.trim()), 300);
   170→    return () => clearTimeout(t);
   171→  }, [query]);
   172→
   173→  const fetchList = useCallback(async () => {
   174→    setLoading(true);
   175→    try {
   176→      const url = `/api/customers${debouncedQuery ? `?q=${encodeURIComponent(debouncedQuery)}` : ""}`;
   177→      const res = await fetch(url, { cache: "no-store" });
   178→      const json = await res.json();
   179→      if (json.error) throw new Error(json.error);
   180→      setCustomers(json.data ?? []);
   181→    } catch (err: any) {
   182→      toast.error(err?.message || "Erro ao carregar clientes");
   183→      setCustomers([]);
   184→    } finally {
   185→      setLoading(false);
   186→    }
   187→  }, [debouncedQuery]);
   188→
   189→  useEffect(() => {
   190→    fetchList();
   191→  }, [fetchList]);
   192→
   193→  const handleCreated = (customer: CustomerListItem) => {
   194→    setCustomers((prev) => [customer, ...prev]);
   195→    setCreateOpen(false);
   196→    toast.success("Cliente cadastrado com sucesso");
   197→  };

   199→  const handleUpdated = (updated: CustomerListItem) => {
   200→    setCustomers((prev) =>
   201→      prev.map((c) => (c.id === updated.id ? { ...c, ...updated } : c))
   202→    );

   204→    if (selectedId === updated.id) {
   205→      // detail will be re-fetched by the DetailPanel via key change
   206→      setSelectedId(null);
   207→      setTimeout(() => setSelectedId(updated.id), 50);
   208→    }
   209→  };
   210→
   211→  const handleDeleted = (id: string) => {
   212→    setCustomers((prev) => prev.filter((c) => c.id !== id));
   213→    setSelectedId(null);
   214→    toast.success("Cliente excluído");
   215→  };
   216→
   217→  return (
   218→    <div>
   219→      <ViewHeader
   220→        title="Cadastro de Clientes"

   222→        icon={<Users className="h-5 w-5" />}
   223→        accent="orange"
   224→        actions={
   225→          <Button
   226→            onClick={() => setCreateOpen(true)}
   227→            className="gradient-orange text-primary-foreground glow-orange"
   228→            size="sm"
   229→          >
   230→            <UserPlus className="h-4 w-4" />
   231→            <span className="hidden sm:inline">Novo cliente</span>
   232→            <span className="sm:hidden">Novo</span>
   233→          </Button>
   234→        }
   235→      />
   236→
   237→      {/* Search bar */}
   238→      <SectionCard className="mb-4">
   239→        <div className="relative">
   240→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   241→          <Input
   242→            value={query}
   243→            onChange={(e) => setQuery(e.target.value)}
   244→            placeholder="Buscar por nome, telefone, CPF/CNPJ ou e-mail..."
   245→            className="h-10 rounded-xl border-border bg-background/50 pl-9"
   246→          />
   247→        </div>
   248→        {!loading && (
   249→          <p className="mt-2 text-xs text-muted-foreground">
   250→            {customers.length} cliente{customers.length === 1 ? "" : "s"}
   251→            {debouncedQuery ? ` para "${debouncedQuery}"` : ""}
   252→          </p>
   253→        )}
   254→      </SectionCard>
   255→
   256→      {/* Customer list */}
   257→      {loading ? (
   258→        <SectionCard>
   259→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   260→            {Array.from({ length: 6 }).map((_, i) => (
   261→              <CustomerCardSkeleton key={i} />
   262→            ))}
   263→          </div>
   264→        </SectionCard>
   265→      ) : customers.length === 0 ? (
   266→        <SectionCard>
   267→          <EmptyState
   268→            icon={<Users className="h-10 w-10 opacity-40" />}
   269→            title={debouncedQuery ? "Nenhum cliente encontrado" : "Nenhum cliente cadastrado"}
   270→            description={
   271→              debouncedQuery
   272→                ? "Tente outro termo de busca."
   273→                : "Comece cadastrando seu primeiro cliente."
   274→            }
   275→            action={
   276→              !debouncedQuery && (
   277→                <Button
   278→                  onClick={() => setCreateOpen(true)}
   279→                  className="gradient-orange text-primary-foreground glow-orange"
   280→                  size="sm"
   281→                >
   282→                  <UserPlus className="h-4 w-4" /> Novo cliente
   283→                </Button>
   284→              )
   285→            }
   286→          />
   287→        </SectionCard>
   288→      ) : (
   289→        <SectionCard className="p-0 sm:p-0">
   290→          <div className="grid gap-3 p-4 sm:grid-cols-2 lg:grid-cols-3 sm:p-5">
   291→            {customers.map((c, i) => (
   292→              <motion.button
   293→                key={c.id}
   294→                onClick={() => setSelectedId(c.id)}
   295→                initial={{ opacity: 0, y: 6 }}
   296→                animate={{ opacity: 1, y: 0 }}
   297→                transition={{ delay: Math.min(i * 0.02, 0.2) }}
   298→                whileHover={{ y: -2 }}
   299→                className="card-premium group flex flex-col gap-3 rounded-2xl border border-border bg-card/50 p-4 text-left backdrop-blur-sm"
   300→              >
   301→                <div className="flex items-start gap-3">
   302→                  <div className="grid h-11 w-11 shrink-0 place-items-center rounded-xl gradient-orange text-sm font-bold text-primary-foreground">
   303→                    {getInitials(c.name)}
   304→                  </div>
   305→                  <div className="min-w-0 flex-1">
   306→                    <p className="truncate text-sm font-semibold text-foreground">
   307→                      {c.name}
   308→                    </p>
   309→                    <p className="mt-0.5 flex items-center gap-1 truncate text-[11px] text-muted-foreground">
   310→                      <Phone className="h-3 w-3" />
   311→                      {c.phone || "Sem telefone"}
   312→                    </p>
   313→                  </div>
   314→                  <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground transition group-hover:text-orange-400" />
   315→                </div>
   316→
   317→                <div className="grid grid-cols-2 gap-2 text-[11px]">
   318→                  <div className="flex items-center gap-1 rounded-md bg-background/40 px-2 py-1">
   319→                    <IdCard className="h-3 w-3 text-blue-300" />
   320→                    <span className="truncate text-muted-foreground">
   321→                      {c.document || "Sem CPF"}
   322→                    </span>
   323→                  </div>
   324→                  <div className="flex items-center gap-1 rounded-md bg-background/40 px-2 py-1">
   325→                    <Car className="h-3 w-3 text-orange-400" />
   326→                    <span className="text-muted-foreground">
   327→                      {c.vehicles.length} veíc.
   328→                    </span>
   329→                  </div>
   330→                </div>
   331→
   332→                <div className="flex flex-wrap items-center gap-1.5">
   333→                  <Badge
   334→                    variant="outline"
   335→                    className="border-border text-[10px] font-normal text-muted-foreground"
   336→                  >
   337→                    <Wrench className="mr-0.5 h-3 w-3" />
   338→                    {c._count?.services ?? 0} OS
   339→                  </Badge>
   340→                  <Badge
   341→                    variant="outline"
   342→                    className="border-border text-[10px] font-normal text-muted-foreground"
   343→                  >
   344→                    <FileText className="mr-0.5 h-3 w-3" />
   345→                    {c._count?.quotes ?? 0} orç.
   346→                  </Badge>
   347→                  <span className="ml-auto text-[10px] text-muted-foreground">
   348→                    {formatShortDate(c.createdAt)}
   349→                  </span>
   350→                </div>
   351→              </motion.button>
   352→            ))}
   353→          </div>
   354→        </SectionCard>
   355→      )}
   356→
   357→      {/* Create dialog */}
   358→      <CustomerFormDialog
   359→        open={createOpen}
   360→        onOpenChange={setCreateOpen}

   364→      {/* Detail panel */}
   365→      <CustomerDetailPanel
   366→        customerId={selectedId}
   367→        onClose={() => setSelectedId(null)}
   368→        onUpdated={handleUpdated}
   369→        onDeleted={handleDeleted}
   370→      />
   371→    </div>
   372→  );
   373→}

   423→function CustomerFormDialog({
   424→  open,
   425→  onOpenChange,
   426→  onSaved,
   427→  customer,
   428→}: {
   429→  open: boolean;
   430→  onOpenChange: (v: boolean) => void;
   431→  onSaved?: (c: CustomerListItem) => void;
   432→  customer?: CustomerListItem | null;
   433→}) {
   434→  const [form, setForm] = useState<CustomerFormData>(emptyForm);
   435→  const [saving, setSaving] = useState(false);
   436→  const isEdit = !!customer;
   437→
   438→  useEffect(() => {
   439→    if (open) {
   440→      setForm(
   441→        customer
   442→          ? {
   443→              name: customer.name || "",
   444→              phone: customer.phone || "",
   445→              whatsapp: customer.whatsapp || "",
   446→              email: customer.email || "",
   447→              document: customer.document || "",
   448→              address: customer.address || "",
   449→              notes: customer.notes || "",
   450→            }
   451→          : emptyForm
   452→      );
   453→    }
   454→  }, [open, customer]);
   455→
   456→  const set = (k: keyof CustomerFormData, v: string) =>
   457→    setForm((prev) => ({ ...prev, [k]: v }));
   458→
   459→  const submit = async () => {
   460→    if (!form.name.trim()) {
   461→      toast.error("Nome é obrigatório");
   462→      return;
   463→    }
   464→    if (!form.phone.trim()) {
   465→      toast.error("Telefone é obrigatório");
   466→      return;
   467→    }
   468→    setSaving(true);
   469→    try {
   470→      const url = isEdit
   471→        ? `/api/customers/${customer!.id}`
   472→        : "/api/customers";
   473→      const res = await fetch(url, {
   474→        method: isEdit ? "PUT" : "POST",
   475→        headers: { "Content-Type": "application/json" },
   476→        body: JSON.stringify(form),
   477→      });
   478→      const json = await res.json();
   479→      if (json.error) throw new Error(json.error);
   480→      onSaved?.(json.data);
   481→      onOpenChange(false);
   482→      toast.success(isEdit ? "Cliente atualizado" : "Cliente cadastrado");
   483→    } catch (err: any) {
   484→      toast.error(err?.message || "Erro ao salvar cliente");
   485→    } finally {
   486→      setSaving(false);
   487→    }
   488→  };
   489→
   490→  return (
   491→    <Dialog open={open} onOpenChange={onOpenChange}>
   492→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-premium sm:max-w-2xl">
   493→        <DialogHeader>
   494→          <DialogTitle className="flex items-center gap-2">
   495→            <UserPlus className="h-5 w-5 text-orange-400" />
   496→            {isEdit ? "Editar cliente" : "Novo cliente"}
   497→          </DialogTitle>
   498→          <DialogDescription>
   499→            {isEdit
   500→              ? "Atualize os dados do cliente. Campos com * são obrigatórios."
   501→              : "Preencha os dados do cliente. Campos com * são obrigatórios."}
   502→          </DialogDescription>
   503→        </DialogHeader>
   504→
   505→        <div className="grid gap-4 sm:grid-cols-2">
   506→          <Field label="Nome *" className="sm:col-span-2">
   507→            <Input
   508→              value={form.name}
   509→              onChange={(e) => set("name", e.target.value)}
   510→              placeholder="Nome completo ou razão social"
   511→              className="border-border bg-background/50"
   512→              autoFocus
   513→            />
   514→          </Field>
   515→
   516→          <Field label="Telefone *">
   517→            <div className="relative">
   518→              <Phone className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   519→              <Input
   520→                value={form.phone}
   521→                onChange={(e) => set("phone", e.target.value)}
   522→                placeholder="(11) 99999-9999"
   523→                className="border-border bg-background/50 pl-9"
   524→              />
   525→            </div>
   526→          </Field>
   527→
   528→          <Field label="WhatsApp">
   529→            <div className="relative">
   530→              <MessageCircle className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   531→              <Input
   532→                value={form.whatsapp}
   533→                onChange={(e) => set("whatsapp", e.target.value)}
   534→                placeholder="Mesmo do telefone ou outro"
   535→                className="border-border bg-background/50 pl-9"
   536→              />
   537→            </div>
   538→          </Field>
   539→
   540→          <Field label="E-mail">
   541→            <div className="relative">
   542→              <Mail className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   543→              <Input
   544→                value={form.email}
   545→                onChange={(e) => set("email", e.target.value)}
   546→                placeholder="cliente@email.com"
   547→                type="email"
   548→                className="border-border bg-background/50 pl-9"
   549→              />
   550→            </div>
   551→          </Field>
   552→
   553→          <Field label="CPF / CNPJ">
   554→            <div className="relative">
   555→              <IdCard className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   556→              <Input
   557→                value={form.document}
   558→                onChange={(e) => set("document", e.target.value)}
   559→                placeholder="000.000.000-00"
   560→                className="border-border bg-background/50 pl-9"
   561→              />
   562→            </div>
   563→          </Field>
   564→
   565→          <Field label="Endereço" className="sm:col-span-2">
   566→            <div className="relative">
   567→              <MapPin className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   568→              <Input
   569→                value={form.address}
   570→                onChange={(e) => set("address", e.target.value)}
   571→                placeholder="Rua, número, bairro, cidade"
   572→                className="border-border bg-background/50 pl-9"
   573→              />
   574→            </div>
   575→          </Field>
   576→
   577→          <Field label="Observações" className="sm:col-span-2">
   578→            <Textarea
   579→              value={form.notes}
   580→              onChange={(e) => set("notes", e.target.value)}
   581→              placeholder="Preferências, histórico, alertas..."
   582→              className="min-h-20 resize-y border-border bg-background/50"
   583→            />
   584→          </Field>
   585→        </div>
   586→
   587→        <DialogFooter className="gap-2">
   588→          <Button
   589→            variant="outline"
   590→            onClick={() => onOpenChange(false)}
   591→            className="border-border"
   592→            disabled={saving}
   593→          >
   594→            Cancelar
   595→          </Button>
   596→          <Button
   597→            onClick={submit}
   598→            disabled={saving}
   599→            className="gradient-orange text-primary-foreground glow-orange"
   600→          >
   601→            {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   602→            {isEdit ? "Salvar alterações" : "Cadastrar cliente"}
   603→          </Button>
   604→        </DialogFooter>
   605→      </DialogContent>
   606→    </Dialog>
   607→  );
   608→}

   610→function Field({
   611→  label,
   612→  children,
   613→  className,
   614→}: {
   615→  label: string;
   616→  children: React.ReactNode;
   617→  className?: string;
